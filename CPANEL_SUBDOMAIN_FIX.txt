FIX #3: CREATE .htaccess FOR PROPER ROUTING
============================================

Create /home/qymwtpra/new.kindcommoditiesltd.com/.htaccess

Content:
--------
# Redirect to Frontend directory
DirectoryIndex Frontend/index.php

<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /
    
    # Redirect root to Frontend
    RewriteRule ^$ Frontend/index.php [L]
    
    # Allow direct access to Backend API
    RewriteRule ^Backend/ - [L]
    
    # Allow direct access to Frontend
    RewriteRule ^Frontend/ - [L]
    
    # Security: Block access to sensitive files
    RewriteRule ^\.git - [F,L]
    RewriteRule ^\.env - [F,L]
    RewriteRule ^setup_database\.php - [F,L]
</IfModule>

# Enable PHP error display (ONLY for testing - disable in production)
# php_flag display_errors On
# php_value error_reporting E_ALL

# Set default charset
AddDefaultCharset UTF-8

# Disable directory browsing
Options -Indexes

# Block access to configuration files
<FilesMatch "\.(env|json|lock|md|txt)$">
    Order allow,deny
    Deny from all
</FilesMatch>
